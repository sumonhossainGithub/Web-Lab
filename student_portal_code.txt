// File: app/Models/Student.php
class Student extends Model
{
    protected $fillable = ['name', 'roll', 'department', 'email'];

    public function results(): HasMany
    {
        return $this->hasMany(Result::class);
    }

    public function getGpaAttribute(): float
    {
        if ($this->results->isEmpty()) return 0.00;
        $points = $this->results->sum(fn($r) => Result::calculateGradePoint((float)$r->marks));
        return round($points / $this->results->count(), 2);
    }
}

// File: app/Models/Result.php
class Result extends Model
{
    protected $fillable = ['student_id', 'course_code', 'marks', 'grade'];

    public function student(): BelongsTo
    {
        return $this->belongsTo(Student::class);
    }

    public static function calculateGrade(float $marks): string
    {
        if ($marks >= 80) return 'A+';
        if ($marks >= 75) return 'A';
        if ($marks >= 70) return 'A-';
        if ($marks >= 65) return 'B+';
        if ($marks >= 60) return 'B';
        if ($marks >= 55) return 'B-';
        if ($marks >= 50) return 'C+';
        if ($marks >= 45) return 'C';
        if ($marks >= 40) return 'D';
        return 'F';
    }

    public static function calculateGradePoint(float $marks): float
    {
        if ($marks >= 80) return 4.00;
        if ($marks >= 75) return 3.75;
        if ($marks >= 70) return 3.50;
        if ($marks >= 65) return 3.25;
        if ($marks >= 60) return 3.00;
        if ($marks >= 55) return 2.75;
        if ($marks >= 50) return 2.50;
        if ($marks >= 45) return 2.25;
        if ($marks >= 40) return 2.00;
        return 0.00;
    }
}


3. RESOURCE CONTROLLER (CRUD)
--------------------------------------------------------------------------------
// File: app/Http/Controllers/StudentController.php
namespace App\Http\Controllers;

use App\Models\Student;
use App\Http\Requests\StoreStudentRequest;
use App\Http\Requests\UpdateStudentRequest;
use Illuminate\Http\Request;

class StudentController extends Controller
{
    // 1. READ ALL (Search & Paginate)
    public function index(Request $request)
    {
        $query = Student::withCount('results');
        if ($search = $request->input('search')) {
            $query->where('name', 'like', "%$search%")
                  ->orWhere('roll', 'like', "%$search%")
                  ->orWhere('department', 'like', "%$search%");
        }
        $students = $query->latest()->paginate(8);
        return view('students.index', compact('students'));
    }

    // 2. CREATE FORM
    public function create()
    {
        return view('students.create');
    }

    // 3. STORE (Create in DB with Form Request Validation)
    public function store(StoreStudentRequest $request)
    {
        $student = Student::create($request->validated());
        return redirect()->route('students.show', $student)
            ->with('success', "Student '{$student->name}' registered successfully!");
    }

    // 4. READ ONE (Show Details & Calculated Results)
    public function show(Student $student)
    {
        $student->load('results');
        return view('students.show', compact('student'));
    }

    // 5. EDIT FORM
    public function edit(Student $student)
    {
        return view('students.edit', compact('student'));
    }

    // 6. UPDATE (Update in DB with Validation)
    public function update(UpdateStudentRequest $request, Student $student)
    {
        $student->update($request->validated());
        return redirect()->route('students.show', $student)
            ->with('success', "Student record updated!");
    }

    // 7. DESTROY (Delete from DB)
    public function destroy(Student $student)
    {
        $student->delete();
        return redirect()->route('students.index')
            ->with('success', "Student record deleted.");
    }
}


4. RESULT CONTROLLER (AUTO-CALCULATES GRADE)
--------------------------------------------------------------------------------
// File: app/Http/Controllers/ResultController.php
namespace App\Http\Controllers;

use App\Models\Student;
use App\Models\Result;
use App\Http\Requests\StoreResultRequest;

class ResultController extends Controller
{
    public function store(StoreResultRequest $request, Student $student)
    {
        $marks = (float) $request->validated('marks');
        $grade = Result::calculateGrade($marks); // Auto-calculated

        $student->results()->create([
            'course_code' => strtoupper($request->validated('course_code')),
            'marks'       => $marks,
            'grade'       => $grade,
        ]);

        return redirect()->route('students.show', $student)
            ->with('success', "Result saved with Grade: $grade!");
    }

    public function destroy(Student $student, Result $result)
    {
        $result->delete();
        return redirect()->route('students.show', $student)->with('success', 'Result removed.');
    }
}

use App\Http\Controllers\StudentController;
use App\Http\Controllers\ResultController;

// Public Routes
Route::get('/students', [StudentController::class, 'index'])->name('students.index');
Route::get('/students/{student}', [StudentController::class, 'show'])->name('students.show');

// Protected Routes (Auth Middleware)
Route::middleware(['auth'])->group(function () {
    Route::get('/students/create', [StudentController::class, 'create'])->name('students.create');
    Route::post('/students', [StudentController::class, 'store'])->name('students.store');
    Route::get('/students/{student}/edit', [StudentController::class, 'edit'])->name('students.edit');
    Route::put('/students/{student}', [StudentController::class, 'update'])->name('students.update');
    Route::delete('/students/{student}', [StudentController::class, 'destroy'])->name('students.destroy');

    Route::post('/students/{student}/results', [ResultController::class, 'store'])->name('results.store');
    Route::delete('/students/{student}/results/{result}', [ResultController::class, 'destroy'])->name('results.destroy');
});
